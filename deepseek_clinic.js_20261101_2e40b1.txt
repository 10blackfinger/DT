   ============================================================ */

const TOOTH_CROWN_COLOR = 0xf5f0e6;
const TOOTH_ROOT_COLOR = 0xe3d9c6;
const TOOTH_DEFAULT_HIGHLIGHT = 0x22c55e;
const TOOTH_LIFT_AMOUNT = 0.3;
const TOOTH_ANIM_DURATION = 500;

/* ============================================================
   أدوات داخلية مساعدة (غير مصدّرة)
   ============================================================ */

function createToothSurfaceMaterial(color) {
  return new THREE.MeshStandardMaterial({
    color: color,
    roughness: 0.38,
    metalness: 0.04
  });
}

function buildSingleTooth(toothId, isUpper) {
  const toothGroup = new THREE.Group();
  toothGroup.name = 'tooth_' + toothId;

  const crownMaterial = createToothSurfaceMaterial(TOOTH_CROWN_COLOR);
  const rootMaterial = createToothSurfaceMaterial(TOOTH_ROOT_COLOR);

  const crownMesh = new THREE.Mesh(
    new THREE.CylinderGeometry(0.065, 0.075, 0.15, 12),
    crownMaterial
  );
  crownMesh.castShadow = true;
  crownMesh.receiveShadow = true;
  toothGroup.add(crownMesh);

  const rootMesh = new THREE.Mesh(
    new THREE.ConeGeometry(0.055, 0.2, 12),
    rootMaterial
  );
  rootMesh.castShadow = true;
  rootMesh.receiveShadow = true;

  if (isUpper) {
    rootMesh.rotation.x = 0;
    rootMesh.position.y = 0.17;
  } else {
    rootMesh.rotation.x = Math.PI;
    rootMesh.position.y = -0.17;
  }
  toothGroup.add(rootMesh);

  toothGroup.userData = {
    type: 'tooth',
    id: toothId,
    name: TOOTH_NAMES[toothId] || ('السن ' + toothId)
  };

  return toothGroup;
}

function findToothByObject(root, toothId) {
  let found = null;
  if (!root || typeof root.traverse !== 'function') return null;

  root.traverse(function (object) {
    if (found) return;
    const data = object.userData;
    if (data && data.type === 'tooth' && String(data.id) === String(toothId)) {
      found = object;
    }
  });

  return found;
}

function eachToothMesh(toothObject, callback) {
  toothObject.traverse(function (child) {
    if (child.isMesh && child.material) {
      const materials = Array.isArray(child.material) ? child.material : [child.material];
      for (let i = 0; i < materials.length; i++) {
        callback(child, materials[i]);
      }
    }
  });
}

/* ============================================================
   2) createToothSet — 32 سن في قوسين
   ============================================================ */

export function createToothSet(scene) {
  const teethGroup = new THREE.Group();
  teethGroup.name = 'teethGroup';

  const ANGLE_STEP = 1.34 / 8;

  /* ----- القوس العلوي (y = 1.5) ----- */
  /* الفهرس 0..7  => 11..18 على يمين المريض (x سالب) */
  /* الفهرس 8..15 => 21..28 على يسار المريض (x موجب)  */
  const upperIds = [
    '11', '12', '13', '14', '15', '16', '17', '18',
    '21', '22', '23', '24', '25', '26', '27', '28'
  ];
  const upperY = 1.5;
  const upperRadiusX = 1.15;
  const upperRadiusZ = 1.0;
  const upperZOffset = 0.55;

  for (let upperIndex = 0; upperIndex < upperIds.length; upperIndex++) {
    const angle = (upperIndex < 8)
      ? -(upperIndex + 1) * ANGLE_STEP
      : (upperIndex - 7) * ANGLE_STEP;

    const posX = upperRadiusX * Math.sin(angle);
    const posZ = upperRadiusZ * Math.cos(angle) - upperZOffset;

    const upperTooth = buildSingleTooth(upperIds[upperIndex], true);
    upperTooth.position.set(posX, upperY, posZ);
    teethGroup.add(upperTooth);
  }

  /* ----- القوس السفلي (y = 1.2) ----- */
  /* الفهرس 0..7  => 41..48 على يمين المريض (x سالب) */
  /* الفهرس 8..15 => 31..38 على يسار المريض (x موجب)  */
  const lowerIds = [
    '41', '42', '43', '44', '45', '46', '47', '48',
    '31', '32', '33', '34', '35', '36', '37', '38'
  ];
  const lowerY = 1.2;
  const lowerRadiusX = 1.05;
  const lowerRadiusZ = 0.92;
  const lowerZOffset = 0.5;

  for (let lowerIndex = 0; lowerIndex < lowerIds.length; lowerIndex++) {
    const angle = (lowerIndex < 8)
      ? -(lowerIndex + 1) * ANGLE_STEP
      : (lowerIndex - 7) * ANGLE_STEP;

    const posX = lowerRadiusX * Math.sin(angle);
    const posZ = lowerRadiusZ * Math.cos(angle) - lowerZOffset;

    const lowerTooth = buildSingleTooth(lowerIds[lowerIndex], false);
    lowerTooth.position.set(posX, lowerY, posZ);
    teethGroup.add(lowerTooth);
  }

  teethGroup.position.set(0, 0, 0);

  if (scene && scene.add) {
    scene.add(teethGroup);
  }

  return teethGroup;
}

/* ============================================================
   4) highlightTooth
   ============================================================ */

export function highlightTooth(scene, toothId, color) {
  const highlightColor = (color === undefined || color === null)
    ? TOOTH_DEFAULT_HIGHLIGHT
    : color;

  const targetTooth = findToothByObject(scene, toothId);
  if (!targetTooth) return;

  const threeColor = new THREE.Color(highlightColor);
  const hexColor = threeColor.getHex();

  eachToothMesh(targetTooth, function (mesh, material) {
    if (!material || !material.color) return;

    if (mesh.userData.originalColor === undefined) {
      mesh.userData.originalColor = material.color.getHex();
    }

    material.color.setHex(hexColor);

    if (material.emissive) {
      material.emissive.setHex(hexColor);
      material.emissiveIntensity = 0.45;
    }
  });

  targetTooth.userData.highlighted = true;
  targetTooth.userData.highlightColor = hexColor;
}

/* ============================================================
   5) resetTeethHighlights
   ============================================================ */

export function resetTeethHighlights(scene) {
  if (!scene || typeof scene.traverse !== 'function') return;

  scene.traverse(function (object) {
    const data = object.userData;
    if (!data || data.type !== 'tooth') return;

    data.highlighted = false;

    eachToothMesh(object, function (mesh, material) {
      if (!material) return;

      if (mesh.userData.originalColor !== undefined && material.color) {
        material.color.setHex(mesh.userData.originalColor);
      }

      if (material.emissive) {
        material.emissive.setHex(0x000000);
        material.emissiveIntensity = 0;
      }
    });
  });
}

/* ============================================================
   6) animateTooth
   ============================================================ */

export function animateTooth(tooth, direction) {
  if (!tooth || !tooth.position) return;

  if (tooth.userData.originalY === undefined) {
    tooth.userData.originalY = tooth.position.y;
  }

  const originY = tooth.userData.originalY;
  const startY = tooth.position.y;
  const targetY = (direction === 'down') ? originY : originY + TOOTH_LIFT_AMOUNT;

  const animationId = (tooth.userData.animationId || 0) + 1;
  tooth.userData.animationId = animationId;

  const nowFunction = (typeof performance !== 'undefined' && performance.now)
    ? function () { return performance.now(); }
    : function () { return Date.now(); };

  const startTime = nowFunction();

  const step = function () {
    if (tooth.userData.animationId !== animationId) return;

    const elapsed = nowFunction() - startTime;
    let progress = elapsed / TOOTH_ANIM_DURATION;

    if (progress > 1) progress = 1;
    if (progress < 0) progress = 0;

    const eased = (progress < 0.5)
      ? (2 * progress * progress)
      : (-1 + (4 - 2 * progress) * progress);

    tooth.position.y = startY + (targetY - startY) * eased;

    if (progress < 1) {
      requestAnimationFrame(step);
    } else {
      tooth.position.y = targetY;
    }
  };

  requestAnimationFrame(step);
}
